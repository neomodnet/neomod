# the mods of a multiplayer room are only in effect for as long as we are in it: the player's own ones
# stay untouched below them and are back afterwards, exactly. that includes what Replay::Mods has no
# idea about but the mod selector resets along with everything else (mod_fadingcursor, which used to
# be gone after a visit to a room), a custom speed, and mods that get changed from inside the room.
options_save_on_back 0
debug_fixed_frametime 0.0166667
mouse_raw_input 0
mouse_sensitivity 1
ui_scale_to_dpi 0
windowed_resolution 1280x720
mouse_to 10 10
@wait 30
mod_hidden 1
mod_hardrock 1
mod_fadingcursor 1
speed_override 1.2
debug_fake_room
@wait 20
ui_assert screen_active room 1
ui_assert convar mod_hidden 0
ui_assert convar mod_hardrock 0
ui_assert convar mod_fadingcursor 0
ui_assert convar speed_override -1
ui_assert convar mod_no_pausing 1
mod_easy 1
mod_hidden 1
mod_hidden 0
ui_assert convar mod_easy 1
ui_assert convar mod_hidden 0
debug_fake_online 0
@wait 20
ui_assert visible room 0
ui_assert convar mod_hidden 1
ui_assert convar mod_hardrock 1
ui_assert convar mod_fadingcursor 1
ui_assert convar speed_override 1.2
ui_assert convar mod_easy 0
ui_assert convar mod_no_pausing 0
mod_hidden 0
mod_hardrock 0
mod_fadingcursor 0
speed_override -1
exit
