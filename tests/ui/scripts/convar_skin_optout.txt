# skins can come with settings of their own, which the player can't change while such a skin is in use.
# skin_allow_convars ("Let Skins Change Settings" in the skin options) is the way out: turning it off
# takes what the skin has set away right there (here: while the skin is still loading, which is when the
# reload that comes with it doesn't happen), the
# player's own values are what counts from then on, a skin that gets loaded in the meantime doesn't get
# to set anything, and turning it back on gives the skin in use its settings back.
# (a skin change is ignored while another skin is still loading, hence the @sleep/@wait pairs)
# uitest-datadir: fresh
# uitest-file: skins/cvtest/skin.ini|[General]|Name: cvtest|Author: uitest||[neomod]|songbrowser_button_active_color_a: 100
options_save_on_back 0
debug_fixed_frametime 0.0166667
mouse_raw_input 0
mouse_sensitivity 1
ui_scale_to_dpi 0
windowed_resolution 1280x720
mouse_to 10 10
@wait 30
skin cvtest
ui_assert convar songbrowser_button_active_color_a 100
skin_allow_convars 0
ui_assert convar songbrowser_button_active_color_a 230
@sleep 500
@wait 30
@sleep 500
@wait 30
ui_assert convar songbrowser_button_active_color_a 230
songbrowser_button_active_color_a 200
ui_assert convar songbrowser_button_active_color_a 200
skin default
@sleep 500
@wait 30
@sleep 500
@wait 30
skin cvtest
@sleep 500
@wait 30
@sleep 500
@wait 30
ui_assert convar songbrowser_button_active_color_a 200
skin_allow_convars 1
@sleep 500
@wait 30
@sleep 500
@wait 30
ui_assert convar songbrowser_button_active_color_a 100
exit
