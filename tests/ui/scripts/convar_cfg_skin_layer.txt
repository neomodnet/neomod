# values a skin forces must not end up in osu.cfg: what gets saved is the client's own value, no
# matter what is overriding it at the time. the save used to run on the effective values, so the
# skin's fposu_fov replaced the client's, and hud_scale (forced to what happens to be its default)
# lost its line altogether.
# the save is async (read old cfg -> write), hence the @sleep/@wait pairs around it; the final
# options_save_on_back 0 keeps the exit path from saving again.
# uitest-datadir: fresh
# uitest-file: skins/cvtest/skin.ini|[General]|Name: cvtest|Author: uitest||[neomod]|fposu_fov: 99|hud_scale: 1
# uitest-cfg: fposu_fov 77
# uitest-cfg: hud_scale 1.5
debug_fixed_frametime 0.0166667
mouse_raw_input 0
mouse_sensitivity 1
ui_scale_to_dpi 0
windowed_resolution 1280x720
mouse_to 10 10
@wait 30
@sleep 200
@wait 5
fposu_fov 77
hud_scale 1.5
skin cvtest
@wait 5
ui_assert convar fposu_fov 99
ui_assert convar hud_scale 1
@sleep 500
@wait 30
@sleep 500
@wait 30
set_active_ui_screen optionsoverlay
@wait 20
ui_assert visible optionsoverlay 1
sendkey Escape
@wait 5
@sleep 200
@wait 5
@sleep 200
@wait 5
options_save_on_back 0
exit
