# a replay plays at its own speed also right after a play at exactly DT/HT: the DT/HT buttons' convars (which only
# mirror the speed) went off with the mod selector's reset when the replay's mods were used, and their deferred
# callback undid the replay's speed (replays of anything at 1x after a DT/HT play). here the instant replay (F2 from
# the pause menu) of a DT play on the playable fixture, as in mods_replay_session.
# assertion-only (audio decodes on wall-clock time); the first 'exit' only stops the replay.
options_save_on_back 0
debug_fixed_frametime 0.0166667
ui_scale_to_dpi 0
windowed_resolution 1280x720
mod_nofail 1
mod_doubletime_dummy 1
mouse_to 10 10
@wait 30
osu_folder uitest_osu_folder_play
set_active_ui_screen songbrowser
@wait 300
ui_assert convar speed_override 1.5
sendkey Return
@wait_secs 5
ui_assert visible songbrowser 0
@sleep 4000
@wait 12
sendkey Escape
@wait 12
ui_assert visible pauseoverlay 1
sendkey F2
@wait 5
ui_assert visible pauseoverlay 0
ui_assert convar speed_override 1.5
ui_assert convar mod_doubletime_dummy 1
exit
@sleep 1000
@wait 60
ui_assert visible songbrowser 1
ui_assert convar speed_override 1.5
mod_nofail 0
mod_doubletime_dummy 0
exit
