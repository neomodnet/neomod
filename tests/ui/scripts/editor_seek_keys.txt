# the osu!stable keys and wheel move the editor time (paused, a 500 ms beat, no audio: the time follows seeks only): a
# tick per arrow (Shift: four), timing points on Up/Down, the first and last object on Z and V, Shift+number for the
# divisor, the wheel by ticks and Ctrl+wheel for the divisor.
# uitest-datadir: fresh
# uitest-maps: uitest editor set|-1|2
options_save_on_back 0
debug_fixed_frametime 0.0166667
ui_trace 1
mouse_raw_input 0
mouse_sensitivity 1
ui_scale_to_dpi 0
windowed_resolution 1280x720
mouse_to 10 10
language en
@wait 30
osu_folder uitest_osu_folder
set_active_ui_screen songbrowser
@wait 300
sendtext uitest editor set
@wait 60
mouse_to 1033 327
@wait 2
mouse_down right
@wait 1
mouse_up right
@wait 3
mouse_to 1116 394
@wait 2
mouse_down
@wait 1
mouse_up
@wait 30
mouse_to 640 380
@wait 3
ui_assert text editor_clock 00:00:000
ui_assert text editor_divisor_value 1/1
sendkey Right
@wait 3
ui_assert text editor_clock 00:00:500
sendkey 225 down
sendkey Right
sendkey 225 up
@wait 3
ui_assert text editor_clock 00:02:500
sendkey Left
@wait 3
ui_assert text editor_clock 00:02:000
sendkey Down
@wait 3
ui_assert text editor_clock 00:00:000
sendkey V
@wait 3
ui_assert text editor_clock 00:01:500
sendkey Z
@wait 3
ui_assert text editor_clock 00:01:000
sendkey Z
@wait 3
ui_assert text editor_clock 00:00:000
sendkey 225 down
sendkey 4
sendkey 225 up
@wait 3
ui_assert text editor_divisor_value 1/4
sendkey Right
@wait 3
ui_assert text editor_clock 00:00:125
mouse_wheel -1
@wait 3
ui_assert text editor_clock 00:00:250
mouse_wheel 1
@wait 3
ui_assert text editor_clock 00:00:125
sendkey 224 down
mouse_wheel 1
@wait 3
sendkey 224 up
@wait 3
ui_assert text editor_divisor_value 1/8
exit
